cmake_minimum_required(VERSION 3.10)
project(CadaverNative C)
set(CMAKE_C_STANDARD 99)
add_library(cadaver_game STATIC src/game.c src/crt.c src/intro.c)
if(NOT WIN32)
  target_link_libraries(cadaver_game m)
endif()
add_executable(dump src/dump_main.c)
target_link_libraries(dump cadaver_game)
if(WIN32)
  add_executable(cadaver WIN32 src/win_main.c src/d3d_present.c)
  target_link_libraries(cadaver cadaver_game winmm gdi32 user32 d3d11 dxgi d3dcompiler dxguid)
endif()
